1function [newbox,nextcube] = place(box, cube, transno, rotvec)
 2% % Place a piece given current box, target cube and valid rotation vector
 3% % (choose first rotation)
 4% 
 5% global P RM
 6% 
 7% piece = P(:,:,transno);         % only deal with this translation
 8% 
 9% % find 1's based coordinate of empty cube
10% [xi,yi,zi] = ind2sub3d(length(box),cube);
11% cubeoffset = repmat([xi-1;yi-1;zi-1],1,length(piece)); % find zero-based translation of cube
12% % place first rotation
13% Ri = rotvec(1);
14% pr = cubeoffset + RM(:,:,Ri)*piece;
15% [filled,newbox] = boxit(box,pr);
16% 
17% 
18% % find next free cube in linear space
19% boxlinear = newbox(:);             % turn box into a linear array with ones at filled locs
20% for nextemptyind=1:numel(newbox)      % scan through empty cubes in turn
21%     if ~boxlinear(nextemptyind)
22%         %disp(nextempty)
23%         break;                     % found next empty cube
24%     end
25% end
26% nextcube = nextemptyind;        % record this target cube
27nextcube = cube+1;
28newbox = box;
