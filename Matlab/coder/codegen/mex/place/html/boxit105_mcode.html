1function [status,box] = boxit(box,p)
 2% Check if piece will clash if inserted into box, insert if it does
 3% piece is zero-based coords
 4
 5% convert p's cube coordinates to linear indicies of box
 6%boxidx=sub2ind(size(box),p(1,:)+1,p(2,:)+1,p(3,:)+1);
 7
 8%blen = length(box);
 9%boxidx = blen^2*p(3,:) + blen*p(2,:) + p(1,:)+1;   % fast version
10
11boxidx = 25*p(3,:) + 5*p(2,:) + p(1,:)+1;   % faster version
12
13if any(box(boxidx))
14    status = false;
15    return
16else
17    
18    % Insert piece array into box and return true
19    box(boxidx) = ones(1,length(p));
20    status = true;
21end
