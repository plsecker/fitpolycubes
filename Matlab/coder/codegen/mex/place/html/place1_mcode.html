1function [newbox,nextcube] = place(box, cube, transno, rot)
 2% Place a piece into target cube with valid trans and valid rotation
 3% into a known good box
 4
 5global P RM
 6
 7piece = P(:,:,transno);         % only deal with this translation
 8
 9% find 1's based coordinate of empty cube
10[xi,yi,zi] = ind2sub3d(length(box),cube);
11cubeoffset = repmat([xi-1;yi-1;zi-1],1,length(piece)); % find zero-based translation of cube
12% place first rotation
13Ri = rot;
14pr = cubeoffset + RM(:,:,Ri)*piece;
15[~,newbox] = boxit(box,pr);
16
17
18% find next free cube in linear space
19boxlinear = newbox(:);             % turn box into a linear array with ones at filled locs
20for nextemptyind=1:numel(newbox)      % scan through empty cubes in turn
21    if ~boxlinear(nextemptyind)
22        %disp(nextempty)
23        break;                     % found next empty cube
24    end
25end
26nextcube = nextemptyind;        % record this target cube
