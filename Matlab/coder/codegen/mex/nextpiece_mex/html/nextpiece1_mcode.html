1function [nexttrans,nextrotvec] = nextpiece(box, nextcube)
 2% Return next piece translation and valid rotation vector for a cube
 3% (which could be empty)
 4
 5global boxcuberots P RM transvec recursedepth
 6
 7%nexttrans = randperm(5);       % random option
 8
 9if recursedepth > 12
10    
11    %%
12    nexttrans = transvec;           % return new translations to try
13    % extract valid rotations for this cube and 1st translation
14    thiscubesrots = boxcuberots(nextcube,:,nexttrans(1));
15    nextrotvec = find(thiscubesrots);       % get all rots for this cube/translation
16    % if isempty(nextrotvec)                  % possible to have no valid rotations for this cube
17    %     disp('No valid rots available for current cube?');
18    % end
19    
20else
21    
22    %% Choose transvec which fills lowest layers
23    % find 1's based coordinate of empty cube
24    [xi,yi,zi] = ind2sub3d(length(box),nextcube);
25    nextcubeoffset = repmat([xi-1;yi-1;zi-1],1,length(P)); % find zero-based translation of cube
26    
27    lowesttranssum = inf(1,5);
28    lowestrotvec = -1;
29    
30    for transno = 1:5
31        thiscubesrots = boxcuberots(nextcube,:,transno);
32        nextrotvec = find(thiscubesrots);       % get rots for this cube/translation
33        piece = P(:,:,transno);                 % only deal with this translation
34        for Ri = nextrotvec
35            pr = nextcubeoffset + RM(:,:,Ri)*piece;
36            boxidx = 25*pr(3,:) + 5*pr(2,:) + pr(1,:)+1;   % find indexes of placed piece
37            status = any(box(boxidx));
38            if ~status
39                indexsum = sum(boxidx);
40                if indexsum < lowesttranssum(transno)
41                    lowesttranssum(transno) = indexsum;
42                    %                lowestrot(transno) = nextrotvec;
43                end
44            end
45        end
46    end
47    
48    valid = isfinite(lowesttranssum);
49    [lowest,nexttrans] = sort(lowesttranssum);
50    valid = isfinite(lowest);
51    nexttrans = nexttrans(valid);           % weed out trans with no fills
52    if ~isempty(nexttrans)
53        thiscubesrots = boxcuberots(nextcube,:,nexttrans(1));
54        nextrotvec = find(thiscubesrots);       % get all rots for this cube/translation
55    else
56        %    nextrotvec = [];
57    end
58    
59end
