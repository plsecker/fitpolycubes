1function [status,box] = boxit(box,p)
 2% Check if piece will clash if inserted into box, insert if it does
 3
 4% convert p's cube coordinates to linear indicies of box
 5%boxidx=sub2ind(size(box),p(1,:)+1,p(2,:)+1,p(3,:)+1);
 6
 7%blen = length(box);
 8%boxidx = blen^2*p(3,:) + blen*p(2,:) + p(1,:)+1;   % fast version
 9
10boxidx = 25*p(3,:) + 5*p(2,:) + p(1,:)+1;   % faster version
11
12if any(box(boxidx))
13    status = false;
14    return
15else
16    
17    % Insert piece array into box and return true
18    box(boxidx) = ones(1,length(p));
19    status = true;
20end
