1function ret = bt(box,cube,transvec,rotvec)
 2
 3global boxcuberots recursedepth
 4ret = 0;
 5
 6if recursedepth <= 4
 7    fprintf(1,'Recurse depth is <= 4: %d\n',int32(recursedepth))
 8end
 9
10recursedepth = recursedepth+1;
11
12%disp('Entering BT with')
13%[cube rotvec]
14if reject(box,cube,transvec,rotvec)
15    %    disp('Leaving BT with same')
16    recursedepth = recursedepth-1;
17    return
18end
19
20% with box, target cube and valid rotvec (1st element) place piece, return
21% new box, new target cube, new piece translations and valid rotations for
22% the first translation
23[box,cube,transvec,rotvec] = place(box,cube,transvec,rotvec);
24
25% monitor
26global boxfilllevel boxfilllevelcount boxmaxfilllevel
27% boxfilllevel(boxfilllevelcount) =  sum(box(:));
28boxfilllevelcount = boxfilllevelcount+1;
29if sum(box(:))/5 > boxmaxfilllevel
30    boxmaxfilllevel = sum(box(:))/5;
31    [boxmaxfilllevel boxfilllevelcount]
32    %box
33    if boxmaxfilllevel == 25        % early exit
34        fprintf(1,'Max num pieces placed: %d, total pieces tried: %d, recurse depth %d\n',int32(boxmaxfilllevel),int32(boxfilllevelcount),int32(recursedepth));
35        box
36        ret = 1;
37        return
38    end
39end
40if ~mod(boxfilllevelcount,10000)
41    %toc
42    fprintf(1,'Max num pieces placed: %d, total pieces tried: %d, recurse depth %d\n',int32(boxmaxfilllevel),int32(boxfilllevelcount),int32(recursedepth));
43    if ~mod(boxfilllevelcount,1000000) && boxmaxfilllevel <24
44        fprintf(1,'Max num pieces placed: %d, total pieces tried: %d, recurse depth %d\n',int32(boxmaxfilllevel),int32(boxfilllevelcount),int32(recursedepth));
45        ret = 1;
46        return
47    end
48end
49
50while ~isempty(transvec)
51    while ~isempty(rotvec)
52        ret = bt(box,cube,transvec(1),rotvec(1));     % use next translation and its rotations
53        if ret
54            fprintf(1,'trans: %d, rot: %d\n',transvec(1),rotvec(1));
55            return;
56        end
57        rotvec = rotvec(2:end);         % choose next rot
58    end
59    transvec = transvec(2:end);         % use next translation
60    % extract valid rotations for this cube and 1st translation
61    if ~isempty(transvec)
62        thiscubesrots = boxcuberots(cube,:,transvec(1));
63        rotvec = find(thiscubesrots);       % get all rots for this cube/translation
64    end
65end
66
67recursedepth = recursedepth-1;
68end
