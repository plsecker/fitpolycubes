1function [newbox,nextcube,nexttrans,nextrotvec] = place(box, cube, transno, rotvec)
 2% Place a piece given current box, target cube and valid rotation vector
 3% (choose first rotation)
 4% Return next target cube in newbox and valid rotation vector for that cube
 5% (which could be empty)
 6
 7global boxcuberots P RM transvec
 8
 9piece = P(:,:,transno);         % only deal with this translation
10
11% find 1's based coordinate of empty cube
12[xi,yi,zi] = ind2sub3d(length(box),cube);
13cubeoffset = repmat([xi-1;yi-1;zi-1],1,length(piece)); % find zero-based translation of piece
14% place first rotation
15Ri = rotvec(1);
16pr = cubeoffset + RM(:,:,Ri)*piece;
17[filled,newbox] = boxit(box,pr);
18
19
20% find next free cube in linear space
21boxlinear = newbox(:);             % turn box into a linear array with ones at filled locs
22for nextemptyind=1:numel(newbox)      % scan through empty cubes in turn
23    if ~boxlinear(nextemptyind)
24        %disp(nextempty)
25        break;                     % found next empty cube
26    end
27end
28nextcube = nextemptyind;        % record this target cube
29
30%transvec = randperm(5);
31nexttrans = transvec;           % return new translations to try
32
33% extract valid rotations for this cube and 1st translation
34thiscubesrots = boxcuberots(nextcube,:,nexttrans(1));
35nextrotvec = find(thiscubesrots);       % get all rots for this cube/translation
36if isempty(nextrotvec)                  % possible to have no valid rotations for this cube
37    %disp('No valid rots available for current cube?');
38end
39
40% check if can reject early based on a single empty cube on layers 3 or 4 or 5
41status = false;
42if sum(sum((newbox(:,:,3)))) == 24
43    status = true;
44% elseif sum(sum((newbox(:,:,4)))) == 24
45%     status = true;
46% elseif sum(sum((newbox(:,:,5)))) == 24
47%     status = true;
48end
49if status == true
50    %nexttrans = [];
51    %nextrotvec = [];
52end
